#version 330 core

// Feixe de luz das provas: um retângulo alto, já montado no mundo pelo Java (virado para a
// câmera), com a coordenada de textura dizendo onde cada ponto fica dentro do feixe.

in vec3 position;   // att 0: canto do retângulo, no mundo
in vec2 textuv;     // att 3: u de 0 a 1 na largura, v de 0 (embaixo) a 1 (em cima)

uniform mat4 view;
uniform mat4 projection;

out vec2 uv;
out float distancia;

void main(void){
	vec4 pv = vec4(position, 1.0) * view;
	gl_Position = pv * projection;
	distancia = length(pv.xyz);
	uv = textuv;
}
