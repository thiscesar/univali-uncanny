#version 330 core

// Terreno da floresta, com alturas e coordenadas em metros. Mistura os materiais pela
// altura e pela inclinação de cada ponto:
//   folhas secas no chão plano, manchas de grama escura (mais comuns nas partes altas),
//   terra exposta nos declives médios, pedra nos declives fortes, lama na beira dos
//   lagos e água, que reflete o céu, a lua e a lanterna. Por cima, o mapa de detalhes (feito
//   no Java) pinta as trilhas de terra batida e muda o chão na mata morta e no pedregal.
// A luz e a neblina vêm do comum.glsl, iguais às dos outros objetos.

#include "comum.glsl"

uniform sampler2D texChao;
uniform sampler2D texGrama;
uniform sampler2D texTerra;
uniform sampler2D texPedra;
uniform sampler2D texAgua;
uniform float tempo;       // segundos, para a água se mexer
uniform mat4 view;         // para levar a normal da água ao espaço da câmera
uniform sampler2D texDetalhes; // r = trilha, g = mata morta, b = pedregal (o mapa inteiro visto de cima)
uniform float tamanhoMapa;     // metros: a coordenada do mapa de detalhes é a posição dividida por isto

in vec3 normalMundo;
in vec3 normalVista;
in vec2 coordTerreno;      // posição na grade, em metros
in float alturaReal;       // altura real em metros (negativa no fundo dos lagos)
in vec3 posVista;
noperspective in vec2 coordAfim;

out vec4 out_Color;

const float ESCALA_TEXTURA = 0.25;  // uma repetição a cada 4 m

// ---------- ruído procedural, para as bordas entre materiais não ficarem retas ----------
float hash(vec2 p){
	return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}
float ruido(vec2 p){
	vec2 i = floor(p);
	vec2 f = fract(p);
	vec2 u = f * f * (3.0 - 2.0 * f);
	return mix(mix(hash(i), hash(i + vec2(1, 0)), u.x),
	           mix(hash(i + vec2(0, 1)), hash(i + vec2(1, 1)), u.x), u.y);
}

float luminancia(vec3 c){
	return dot(c, vec3(0.299, 0.587, 0.114));
}

// Uma leitura só da textura, repetida a cada 4 m. (Uma cópia ampliada misturada por cima
// também se repetia, a cada 17 m, e formava listras vistas de longe.) Quem disfarça a
// repetição são as manchas de brilho em escala grande, mais abaixo.
vec3 amostra(sampler2D t, vec2 c){
	return texture(t, c).rgb;
}

void main(void){
	if(passadaSombra > 0.5){
		out_Color = vec4(1.0); // desenhando o mapa de sombra: só a profundidade importa
		return;
	}
	// no modo PS1, a textura usa a coordenada sem correção de perspectiva: entorta como no console
	vec2 c = (modoPs1 > 0.5 ? coordAfim : coordTerreno) * ESCALA_TEXTURA;
	float inclinacao = 1.0 - normalize(normalMundo).y; // 0 = plano, 1 = parede
	float h = alturaReal;

	vec3 chao  = amostra(texChao, c);
	vec3 grama = amostra(texGrama, c * 1.3);
	vec3 terra = amostra(texTerra, c);
	vec3 pedra = amostra(texPedra, c * 0.7);

	// manchas de grama: ruído em duas escalas, puxado para cima nas partes altas
	float manchas = ruido(coordTerreno * 0.03) * 0.7 + ruido(coordTerreno * 0.11) * 0.3;
	float pesoGrama = smoothstep(0.45, 0.65, manchas + h * 0.02);
	vec3 cor = mix(chao, grama, pesoGrama);

	// declives: terra exposta, depois pedra
	float pesoTerra = smoothstep(0.05, 0.12, inclinacao + (ruido(coordTerreno * 0.2) - 0.5) * 0.04);
	cor = mix(cor, terra, pesoTerra);
	float pesoPedra = smoothstep(0.16, 0.26, inclinacao);
	cor = mix(cor, pedra, pesoPedra);

	// as regiões e as trilhas (o mapa de detalhes é lido com filtro linear: as bordas saem suaves)
	vec3 mapa = texture(texDetalhes, coordTerreno / tamanhoMapa).rgb;
	// mata morta: chão seco e acinzentado, sem grama
	vec3 seco = mix(chao, vec3(luminancia(chao)), 0.55) * 0.9;
	cor = mix(cor, seco, mapa.g * 0.75);
	// pedregal: pedra aflorando em manchas
	cor = mix(cor, pedra * 0.85, mapa.b * 0.6 * smoothstep(0.35, 0.65, ruido(coordTerreno * 0.18)));
	// trilha: terra batida, um pouco mais clara e lisa que o resto
	cor = mix(cor, terra * 1.5 + vec3(0.05, 0.04, 0.025), mapa.r * 0.9 * smoothstep(0.0, 0.25, h));

	// lama: faixa escura e úmida na beira dos lagos (até ~0,6 m acima da água)
	float pesoLama = 1.0 - smoothstep(0.05, 0.6 + ruido(coordTerreno * 0.3) * 0.3, h);
	cor = mix(cor, terra * 0.45, pesoLama);

	// manchas de brilho de dezenas de metros: a repetição do ladrilho (a cada 4 m) some
	cor *= 0.82 + 0.36 * ruido(coordTerreno * 0.035 + vec2(13.1, 7.7));

	vec3 chaoIluminado = iluminar(cor, normalVista, posVista);

	// água escura, mais funda = mais escura, com o detalhe da textura deslizando
	float profundidade = clamp(-h / 4.0, 0.0, 1.0);
	vec3 corAgua = mix(vec3(0.07, 0.11, 0.11), vec3(0.01, 0.03, 0.04), profundidade);
	vec2 c1 = c + vec2(tempo * 0.020, tempo * 0.013);
	vec2 c2 = c * 0.6 - vec2(tempo * 0.012, -tempo * 0.017);
	float detalhe = (luminancia(texture(texAgua, c1).rgb) + luminancia(texture(texAgua, c2).rgb)) * 0.5;
	corAgua *= 0.7 + 0.6 * detalhe;

	// ondinhas: inclinam a normal da água; o reflexo da lanterna treme junto
	vec2 p = coordTerreno;
	float dx = 0.06 * cos(p.x * 1.3 + tempo * 1.5) + 0.03 * cos((p.x + p.y) * 2.1 + tempo * 2.3);
	float dz = 0.06 * cos(p.y * 1.1 + tempo * 1.2) + 0.03 * cos((p.x - p.y) * 1.7 + tempo * 1.9);
	vec3 normalAgua = normalize((vec4(normalize(vec3(-dx, 1.0, -dz)), 0.0) * view).xyz);
	vec3 V = normalize(-posVista); // do ponto até a câmera

	// reflexo do céu noturno (um pouco mais claro e azulado que a neblina). Efeito de Fresnel:
	// olhando de cima a água é escura e transparente; olhando rente, ela vira um espelho
	float fresnel = 0.1 + 0.9 * pow(1.0 - max(dot(normalAgua, V), 0.0), 4.0);
	vec3 ceu = corNeblina * 1.6 + vec3(0.06, 0.08, 0.12); // o céu enevoado refletido (mais fraco que a névoa: água escura)
	vec3 aguaIluminada = mix(iluminar(corAgua, normalAgua, posVista), ceu, fresnel);

	// a lua: uma luz fixa no céu; as ondinhas quebram o reflexo dela em pontos que tremem
	vec3 lua = normalize((vec4(normalize(vec3(-0.35, 0.45, -0.8)), 0.0) * view).xyz);
	vec3 meioLua = normalize(lua + V);
	aguaIluminada += vec3(0.5, 0.58, 0.72) * pow(max(dot(normalAgua, meioLua), 0.0), 50.0) * 1.2;
	aguaIluminada += brilhoLanterna(normalAgua, posVista, 90.0) * 0.9;

	// a margem: uma faixa clara e úmida onde a água encontra a terra, ondulando devagar
	float margem = smoothstep(-0.15, -0.03, h) * (1.0 - smoothstep(-0.03, 0.01, h));
	aguaIluminada += vec3(0.07, 0.08, 0.08) * margem * (0.5 + 0.5 * sin(tempo * 1.7 + ruido(coordTerreno * 0.8) * 6.0));

	// a margem: alturaReal é interpolada dentro do triângulo, então a linha da água sai suave
	float pesoAgua = 1.0 - smoothstep(-0.05, 0.02, h);
	vec3 corFinal = mix(chaoIluminado, aguaIluminada, pesoAgua);

	out_Color = vec4(aplicarNeblina(corFinal, length(posVista), 0.0), 1.0);
}
