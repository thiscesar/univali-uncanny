#version 330 core

// Árvores desenhadas por instancing: o modelo de uma árvore (dois planos cruzados de
// altura 1) é esticado, girado e posicionado com os dados de cada instância.

#include "ps1Vertice.glsl"

in vec3 position;   // att 0: vértice do modelo (x ou z de -0,5 a 0,5; y de 0 a 1)
in vec3 normal;     // att 2: normal do plano
in vec2 textuv;     // att 3: coordenada de textura dentro de uma variação
in vec4 instancia;  // att 4: x, y (chão), z, altura em metros
in vec4 extra;      // att 5: giro em radianos, largura em metros, brilho, número da textura

uniform mat4 view;
uniform mat4 projection;
uniform float tempo;
uniform float numeroTexturas;

out vec2 uv;
out vec3 normalVista;
out vec3 posVista;
out float brilho;

void main(void){
	float c = cos(extra.x), s = sin(extra.x);
	vec3 p = vec3(position.x * extra.y, position.y * instancia.w, position.z * extra.y);
	vec3 girado = vec3(c * p.x - s * p.z, p.y, s * p.x + c * p.z);

	// vento: só a parte de cima balança (quadrático na altura), cada árvore numa fase
	float fase = tempo * 1.3 + instancia.x * 0.31 + instancia.z * 0.17;
	float balanco = sin(fase) * 0.22 * position.y * position.y;
	girado.x += balanco;
	girado.z += balanco * 0.5;

	// a posição já fica no mundo: não há matriz model
	vec4 pv = vec4(girado + instancia.xyz, 1.0) * view;
	gl_Position = tremerPs1(pv * projection);
	posVista = pv.xyz;

	vec3 n = vec3(c * normal.x - s * normal.z, normal.y, s * normal.x + c * normal.z);
	normalVista = (vec4(n, 0.0) * view).xyz;

	uv = vec2((extra.w + textuv.x) / numeroTexturas, textuv.y);
	brilho = extra.z;
}
