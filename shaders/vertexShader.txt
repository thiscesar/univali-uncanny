#version 330 core

in vec3 position; //att 0
in vec3 cor; // att 1
in vec3 normal; // att 2
in vec2 textuv; // att 3

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;

out vec3 color;
out vec3 norm;
out vec2 uv;
out float distancia;
out float alturaMundo;

void main(void){
	mat4 modelview = model*view;
	gl_Position = (vec4(position,1.0)*modelview*projection);

	// distância até a câmera, para a neblina
	vec4 posVista = vec4(position, 1.0) * modelview;
	distancia = length(posVista.xyz);

	// altura do ponto no mundo, para a neblina baixa
	vec4 posMundo = vec4(position, 1.0) * model;
	alturaMundo = posMundo.y;

	// normal girada junto com o objeto (0.0 = ignora a translação)
	norm = (vec4(normal, 0.0) * model).xyz;

	uv = vec2(textuv.y,1-textuv.x);
	color = vec3(cor.x,cor.y,cor.z);
}
