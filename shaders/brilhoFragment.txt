#version 330 core

// Brilho redondo e suave, somado à cena (glBlendFunc(GL_ONE, GL_ONE)): no escuro, aparece
// mesmo sem nenhuma luz. A neblina o enfraquece com menos força que o resto (metade da
// densidade, ou o fator pedido): os olhos e as janelas atravessam a neblina.

#include "comum.glsl"

uniform vec3 corBrilho;
uniform float fatorNeblina;  // quanto da neblina o brilho sofre (0 = o padrão, metade)

in float distancia;

out vec4 out_Color;

void main(void){
	vec2 d = gl_PointCoord * 2.0 - 1.0;   // de -1 a 1 dentro do quadradinho
	float r2 = dot(d, d);
	if(r2 > 1.0){
		discard; // fora do círculo
	}
	float forca = (1.0 - r2) * (1.0 - r2);
	float fator = fatorNeblina > 0.0 ? fatorNeblina : 0.5;
	float neblina = exp(-pow(densidadeNeblina * fator * distancia, 2.0));
	out_Color = vec4(corBrilho * forca * neblina, 1.0);
}
