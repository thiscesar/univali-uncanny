#version 330 core

// Modo PS1: amplia a imagem pequena sem suavizar (cada pixel vira um bloco). Opcionalmente reduz
// as cores a poucos níveis por canal, com pontilhado ordenado (Bayer 4 x 4), como o PlayStation
// fazia para esconder as faixas de cor (niveis < 2 desliga).

uniform sampler2D cena;
uniform vec2 resolucao;  // tamanho da imagem pequena
uniform float niveis;    // níveis por canal (32 = 5 bits, como no console; 0 = sem redução)

in vec2 uv;

out vec4 out_Color;

const int BAYER[16] = int[16](0, 8, 2, 10, 12, 4, 14, 6, 3, 11, 1, 9, 15, 7, 13, 5);

void main(void){
	ivec2 pixel = ivec2(floor(uv * resolucao));          // o pixel da imagem pequena
	vec3 c = texelFetch(cena, pixel, 0).rgb;              // lido sem suavização
	if (niveis < 2.0) {
		out_Color = vec4(c, 1.0);
		return;
	}
	float limiar = (float(BAYER[(pixel.y % 4) * 4 + (pixel.x % 4)]) + 0.5) / 16.0;
	vec3 q = floor(c * (niveis - 1.0) + limiar) / (niveis - 1.0);
	out_Color = vec4(q, 1.0);
}
