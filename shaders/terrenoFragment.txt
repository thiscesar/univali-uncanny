#version 330 core

// Texture splatting: mistura água, areia, grama, pedra e neve
// conforme a altura real e a inclinação de cada ponto.

uniform sampler2D texAgua;
uniform sampler2D texAreia;
uniform sampler2D texGrama;
uniform sampler2D texPedra;
uniform float tempo;   // segundos, para animar a água
uniform mat4 view;     // para o reflexo do sol na água

in vec3 normalMundo;
in vec2 coordTerreno;
in float alturaReal;
in float alturaMundo;
in float distancia;
in vec3 posVista;

out vec4 out_Color;

// ---------- ajustes ----------
const float ESCALA_TEXTURA = 0.125;   // 1 repetição a cada 8 células da grade
const float FIM_AREIA      = 12.0;    // metros
const float INICIO_PEDRA   = 300.0;   // metros (mesmo limite do código original)
const float INICIO_NEVE    = 420.0;   // metros
const vec3  DIR_SOL        = vec3(0.4, 1.0, 0.3);

// neblina (iguais às do fragmentShader.txt, para combinar com o resto da cena)
const vec3  COR_CEU        = vec3(0.55, 0.75, 0.95);
const float DENSIDADE      = 0.006;
const float FORCA_CHAO     = 0.6;
const float ALTURA_DISSIPA = 3.0;

// ---------- ruído procedural, para as bordas não ficarem retas ----------
float hash(vec2 p){
	return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}
float ruido(vec2 p){
	vec2 i = floor(p);
	vec2 f = fract(p);
	vec2 u = f * f * (3.0 - 2.0 * f);
	return mix(mix(hash(i), hash(i + vec2(1, 0)), u.x),
	           mix(hash(i + vec2(0, 1)), hash(i + vec2(1, 1)), u.x), u.y);
}

float luminancia(vec3 c){
	return dot(c, vec3(0.299, 0.587, 0.114));
}

// Lê a textura em duas escalas e mistura: disfarça a repetição do ladrilho.
vec3 amostra(sampler2D t, vec2 c){
	vec3 perto = texture(t, c).rgb;
	vec3 longe = texture(t, c * 0.21 + vec2(0.37, 0.11)).rgb;
	return mix(perto, longe, 0.35);
}

void main(void){
	vec2 c = coordTerreno * ESCALA_TEXTURA;
	vec3 n = normalize(normalMundo);
	vec3 sol = normalize(DIR_SOL);
	float inclinacao = 1.0 - n.y; // 0 = plano, 1 = parede

	// altura "perturbada" pelo ruído: as fronteiras entre materiais ficam irregulares
	float h = alturaReal
	        + (ruido(coordTerreno * 0.05) - 0.5) * 40.0
	        + (ruido(coordTerreno * 0.40) - 0.5) * 10.0;

	// ---------- terra ----------
	vec3 areia = amostra(texAreia, c);
	vec3 grama = amostra(texGrama, c) * vec3(0.85, 0.95, 0.80); // tira um pouco do amarelo
	vec3 pedra = amostra(texPedra, c * 0.7);

	float pesoAreia = 1.0 - smoothstep(FIM_AREIA - 6.0, FIM_AREIA + 6.0, h);
	float pesoPedraAltura = smoothstep(INICIO_PEDRA - 40.0, INICIO_PEDRA + 40.0, h);
	float pesoPedraIncl = smoothstep(0.20, 0.40, inclinacao) * (1.0 - pesoAreia);
	float pesoNeve = smoothstep(INICIO_NEVE - 30.0, INICIO_NEVE + 30.0, h)
	               * (1.0 - smoothstep(0.35, 0.60, inclinacao)); // neve não para em paredão

	vec3 terra = mix(grama, areia, pesoAreia);
	terra = mix(terra, pedra, max(pesoPedraAltura, pesoPedraIncl));
	vec3 neve = vec3(0.92, 0.94, 0.97) * (0.85 + 0.3 * luminancia(pedra));
	terra = mix(terra, neve, pesoNeve);

	float difusa = max(dot(n, sol), 0.0);
	vec3 terraIluminada = terra * (0.35 + 0.65 * difusa);

	// ---------- água ----------
	float profundidade = clamp(-alturaReal / 150.0, 0.0, 1.0);
	vec3 aguaRasa  = vec3(0.10, 0.50, 0.55);
	vec3 aguaFunda = vec3(0.02, 0.14, 0.28);
	vec3 corAgua = mix(aguaRasa, aguaFunda, profundidade);

	// duas camadas da textura deslizando em direções diferentes
	vec2 c1 = c + vec2(tempo * 0.020, tempo * 0.013);
	vec2 c2 = c * 0.6 - vec2(tempo * 0.012, -tempo * 0.017);
	float detalhe = (luminancia(texture(texAgua, c1).rgb) + luminancia(texture(texAgua, c2).rgb)) * 0.5;
	corAgua *= 0.75 + 0.6 * detalhe;

	// ondinhas: inclinam a normal da água com senos que andam com o tempo
	vec2 p = coordTerreno;
	float ondaDiag = cos((p.x + p.y) * 1.7 + tempo * 2.1);
	float dx = 0.05 * cos(p.x * 0.9 + tempo * 1.5) + 0.03 * ondaDiag;
	float dz = 0.05 * cos(p.y * 1.1 + tempo * 1.2) + 0.03 * ondaDiag;
	vec3 nAgua = normalize(vec3(-dx, 1.0, -dz));

	// reflexo do sol (especular) e do céu (Fresnel), calculados no espaço da câmera
	vec3 N = normalize((vec4(nAgua, 0.0) * view).xyz);
	vec3 L = normalize((vec4(sol, 0.0) * view).xyz);
	vec3 V = normalize(-posVista);
	vec3 H = normalize(L + V);
	float brilho = pow(max(dot(N, H), 0.0), 120.0);
	float fresnel = pow(1.0 - max(dot(N, V), 0.0), 3.0);

	corAgua *= 0.6 + 0.4 * max(dot(nAgua, sol), 0.0);
	corAgua = mix(corAgua, COR_CEU, fresnel * 0.6);
	corAgua += vec3(1.0, 0.95, 0.85) * brilho * 0.8;

	// espuma na beira da praia
	float espuma = (1.0 - smoothstep(0.0, 5.0, -alturaReal))
	             * smoothstep(0.55, 0.75, ruido(coordTerreno * 0.8 + tempo * 0.3));
	corAgua = mix(corAgua, vec3(0.95), espuma * 0.5);

	// ---------- junta terra e água ----------
	// alturaReal é interpolada dentro do triângulo, então a linha da costa sai suave
	float pesoAgua = 1.0 - smoothstep(-1.0, 1.0, alturaReal + (ruido(coordTerreno * 0.3) - 0.5) * 3.0);
	vec3 cor = mix(terraIluminada, corAgua, pesoAgua);

	// ---------- neblina ----------
	float fatorAltura = 1.0 + FORCA_CHAO * exp(-max(alturaMundo, 0.0) / ALTURA_DISSIPA);
	float densidade = DENSIDADE * fatorAltura;
	float neblina = 1.0 - exp(-pow(densidade * distancia, 2.0));

	out_Color = vec4(mix(cor, COR_CEU, neblina), 1.0);
}
